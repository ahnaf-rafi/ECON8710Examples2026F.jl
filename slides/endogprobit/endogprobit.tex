\section*{Introduction}

\begin{frame}{Observables and the structural model}
We observe an i.i.d.\ sample of \(\xi = (Y, D, X', Z')'\).
\(Y \in \{0, 1\}\) is the binary outcome,
\(D \in \R\) is a continuous \textbf{endogenous} covariate,
\(X \in \R^{k}\) are exogenous covariates,
\(Z \in \R^{l}\) are excluded instruments, with \(l \geq 1\).
For unobserved \(\epsilon\) and \(\nu\), model is
\begin{equation*}
  \begin{split}
    Y =
    & \, \ind \{\alpha_{0} D + X' \beta_{0} - \epsilon \geq 0\}, \\
    D =
    & \, X' \gamma_{0} + Z' \pi_{0} + \nu = S' \delta_{0} + \nu, \quad
    S = (X', Z')', \quad \delta_{0} = (\gamma_{0}', \pi_{0}')'.
  \end{split}
\end{equation*}
Assume \(\pi \neq \mathbf{0}_{l}\) (rank condition), \(\E [S S']\) is finite and
positive definite, \((\epsilon, \nu) \indep S\) and
\begin{equation*}
  \begin{pmatrix} \epsilon \\ \nu \end{pmatrix}
  \sim \Normal (\mathbf{0}_{2}, \Sigma_{0}), \quad
  \Sigma_{0} =
  \begin{pmatrix}
    1 & \rho_{0} \sqrt{\tau_{0}} \\
    \rho_{0} \sqrt{\tau_{0}} & \tau_{0}
  \end{pmatrix}.
\end{equation*}
\vfill
\begin{itemize}
  \item We require that \(\rho_{0} \in [-1, 1]\).
    With this parametrization, \(\Sigma_{0}\) is a covariance matrix.
    \vfill
  \item When \(\rho_{0} \neq 0\), \(D\) is endogenous since it correlates with
    \(\varepsilon\).
    \vfill
  \item The full vector of unknown parameters is \(\vartheta_{0} = (\alpha_{0},
    \beta_{0}', \gamma_{0}', \pi_{0}', \tau_{0}, \rho_{0})\).
\end{itemize}
\end{frame}

\begin{frame}{Conditional log-likelihood}
Let \(p_{\vartheta} (\cdot | \cdot)\) be the conditional density of \((Y,
D)\) given \(S\) at candidate \(\vartheta\) for \(\vartheta_{0}\).
For \(\tau > 0\) and \(|\rho| < 1\),
\begin{equation*}
  \begin{split}
    p_{\vartheta} (Y, D | S) =
    & \, \frac{\exp \left\{-\frac{(D - S' \delta)^{2}}{2 \tau}\right\}}
    {\sqrt{2 \pi \tau}} \times \Phi \left(
    \frac{\sqrt{\tau} (\alpha D + X' \beta) - \rho (D - S' \delta)}
    {\sqrt{\tau (1 - \rho^{2})}}\right)^{Y} \\
    & \times \Phi \left(
    \frac{\rho (D - S' \delta) - \sqrt{\tau} (\alpha D + X' \beta)}
    {\sqrt{\tau (1 - \rho^{2})}}\right)^{1 - Y}.
  \end{split}
\end{equation*}
Take logarithms:
\begin{equation*}
  \begin{split}
    \log p_{\vartheta} (Y, D | S) =
    & \, -\frac{\log (2 \pi)}{2} - \frac{1}{2} \log \tau
    - \frac{(D - S' \delta)^{2}}{2 \tau} \\
    & + Y \log \Phi \left(
    \frac{\sqrt{\tau} (\alpha D + X' \beta) - \rho (D - S' \delta)}
    {\sqrt{\tau (1 - \rho^{2})}}\right) \\
    & + (1 - Y) \log \Phi \left(
    \frac{\rho (D - S' \delta) - \sqrt{\tau} (\alpha D + X' \beta)}
    {\sqrt{\tau (1 - \rho^{2})}}\right).
  \end{split}
\end{equation*}
\end{frame}

\begin{frame}{Conditional log-likelihood}
We can drop the constant term and use \(2 Y - 1 \in \{-1, 1\}\) to simplify
things a little:
\begin{equation*}
  \ell_{\vartheta} (\xi) = -\frac{1}{2} \log \tau
  - \frac{(D - S' \delta)^{2}}{2 \tau}
  + \log \Phi \left((2 Y - 1)
  \frac{\sqrt{\tau} (\alpha D + X' \beta) - \rho (D - S' \delta)}
  {\sqrt{\tau (1 - \rho^{2})}}\right).
\end{equation*}
This will also help reduce computational load later on.
The conditional log-likelihood is (up to an additive constant) equal to the
sample sum of the above;
the negative average conditional log-likelihood is that multiplied by a factor
of \(-1/n\).
\end{frame}

\begin{frame}{Conditional log-likelihood: index and its gradient}
Let \(\phi\) be the standard normal density.
For \(\tau > 0\) and \(|\rho| < 1\), define
\begin{equation*}
  q = \frac{\alpha D + X' \beta - \rho (D - S' \delta) / \sqrt{\tau}}
  {\sqrt{1 - \rho^{2}}}, \quad
  r = (2 Y - 1) \frac{\phi ((2 Y - 1) q)}{\Phi ((2 Y - 1) q)}.
\end{equation*}
\vfill
Holding the observation fixed, (with \((\delta = (\gamma', \pi')')\)),
\begin{equation*}
  \begin{gathered}
    \partial_{\alpha} q = \frac{D}{\sqrt{1 - \rho^{2}}}, \quad
    \nabla_{\beta} q = \frac{X}{\sqrt{1 - \rho^{2}}}, \quad
    \nabla_{\delta} q = \frac{\rho S}{\sqrt{\tau (1 - \rho^{2})}}, \\
    \partial_{\tau} q =
    \frac{\rho (D - S' \delta)}{2 \tau^{3/2} \sqrt{1 - \rho^{2}}}, \quad
    \partial_{\rho} q =
    \frac{\rho (\alpha D + X' \beta) - (D - S' \delta) / \sqrt{\tau}}
    {(1 - \rho^{2})^{3/2}}.
  \end{gathered}
\end{equation*}
\end{frame}

\begin{frame}{Conditional log-likelihood: gradient}
Since \(\partial_{q} \log \Phi ((2 Y - 1) q) = r\), it follows from the chain
rule that
\begin{equation*}
  \nabla_{\vartheta} \ell_{\vartheta} (\xi) =
  \begin{pmatrix}
    0 \\
    \mathbf{0}_{k} \\
    (D - S' \delta) S / \tau \\
    ((D - S' \delta)^{2} - \tau) / (2 \tau^{2}) \\
    0
  \end{pmatrix}
  + r \nabla_{\vartheta} q.
\end{equation*}
\vfill
For programming, use the gradient of the negative average conditional
log-likelihood:
\begin{equation*}
  \nabla_{\vartheta} \left(-\frac{1}{n} \sum_{i = 1}^{n}
  \ell_{\vartheta} (\xi_{i})\right) = -\frac{1}{n} \sum_{i = 1}^{n}
  \nabla_{\vartheta} \ell_{\vartheta} (\xi_{i}).
\end{equation*}
\end{frame}

\begin{frame}{Conditional log-likelihood: Hessian}
Holding \(Y\) fixed, \(\partial_{q} r = -r (q + r)\).
Therefore
\begin{equation*}
  \nabla_{\vartheta}^{2} \ell_{\vartheta} (\xi) = \nabla_{\vartheta}^{2}
  \left(-\frac{1}{2} \log \tau - \frac{(D - S' \delta)^{2}}{2 \tau}\right) - r
  (q + r) (\nabla_{\vartheta} q) (\nabla_{\vartheta} q)' + r
  \nabla_{\vartheta}^{2} q.
\end{equation*}
In the above,
\begingroup
\renewcommand{\arraystretch}{1.5}
\begin{equation*}
  \nabla_{\vartheta}^{2} \left(-\frac{1}{2} \log \tau
  - \frac{(D - S' \delta)^{2}}{2 \tau}\right) =
  \begin{pmatrix}
    0 & \mathbf{0}' & \mathbf{0}' & 0 & 0 \\
    \mathbf{0} & \mathbf{0} & \mathbf{0} & \mathbf{0} & \mathbf{0} \\
    \mathbf{0} & \mathbf{0} & -\frac{S S'}{\tau}
    & -\frac{(D - S' \delta) S}{\tau^{2}} & \mathbf{0} \\
    0 & \mathbf{0}' & -\frac{(D - S' \delta) S'}{\tau^{2}}
    & \frac{1}{2 \tau^{2}} - \frac{(D - S' \delta)^{2}}{\tau^{3}} & 0 \\
    0 & \mathbf{0}' & \mathbf{0}' & 0 & 0
  \end{pmatrix}.
\end{equation*}
\endgroup
\end{frame}

\begin{frame}{Conditional log-likelihood: index Hessian}
For fixed \((\tau, \rho)\), the index \(q\) is affine in
\((\alpha, \beta', \delta')'\).
Hence all second derivatives involving only these coefficients are zero.
\vfill
The cross-partial derivatives with respect to \(\tau\) are
\begin{equation*}
  \partial_{\alpha,\tau}^{2} q = 0, \quad
  \nabla_{\beta,\tau}^{2} q = \mathbf{0}_{k}, \quad
  \nabla_{\delta,\tau}^{2} q =
  -\frac{\rho S}{2 \tau^{3/2} \sqrt{1 - \rho^{2}}}.
\end{equation*}
\vfill
The cross-partial derivatives with respect to \(\rho\) are
\begin{equation*}
  \partial_{\alpha,\rho}^{2} q = \frac{\rho D}{(1 - \rho^{2})^{3/2}}, \quad
  \nabla_{\beta,\rho}^{2} q = \frac{\rho X}{(1 - \rho^{2})^{3/2}}, \quad
  \nabla_{\delta,\rho}^{2} q =
  \frac{S}{\sqrt{\tau} (1 - \rho^{2})^{3/2}}.
\end{equation*}
\end{frame}

\begin{frame}{Conditional log-likelihood: index Hessian (continued)}
Remaining second derivatives of \(q\) are
\begin{equation*}
  \begin{gathered}
    \partial_{\tau,\tau}^{2} q =
    -\frac{3 \rho (D - S' \delta)}{4 \tau^{5/2} \sqrt{1 - \rho^{2}}}, \quad
    \partial_{\tau,\rho}^{2} q =
    \frac{D - S' \delta}{2 \tau^{3/2} (1 - \rho^{2})^{3/2}}, \\
    \partial_{\rho,\rho}^{2} q =
    \frac{(1 + 2 \rho^{2}) (\alpha D + X' \beta)
    - 3 \rho (D - S' \delta) / \sqrt{\tau}}{(1 - \rho^{2})^{5/2}}.
  \end{gathered}
\end{equation*}
\vfill
These blocks, together with the previous slide, determine
\(\nabla_{\vartheta}^{2} q\) and hence
\(\nabla_{\vartheta}^{2} \ell_{\vartheta} (\xi)\).
\vfill
For the negative average conditional log-likelihood, the Hessian is
\begin{equation*}
  \nabla_{\vartheta}^{2} \left(-\frac{1}{n} \sum_{i = 1}^{n}
  \ell_{\vartheta} (\xi_{i})\right) = -\frac{1}{n} \sum_{i = 1}^{n}
  \nabla_{\vartheta}^{2} \ell_{\vartheta} (\xi_{i}).
\end{equation*}
\end{frame}

\begin{frame}{Two-step GMM: motivation}
Maximum likelihood estimates all parameters jointly.
The model also suggests a simpler sequence of estimation problems.
\vfill
Equation for \(D\) implies
\begin{equation*}
  \E [S (D - S' \delta_{0})] = \mathbf{0}_{k + l}, \quad
  \E [(D - S' \delta_{0})^{2}] = \tau_{0}.
\end{equation*}
First equation identifies \(\delta_{0}\), and the second identifies
\(\tau_{0}\).
Their sample counterparts give least squares and the average squared residual.
\vfill
For second stage error, conditional on \((\nu, S)\),
\begin{equation*}
  \epsilon \mid (\nu, S) \sim
  \Normal \left(\frac{\rho_{0}}{\sqrt{\tau_{0}}} \nu,
  1 - \rho_{0}^{2}\right).
\end{equation*}
This suggests a probit regression of \(Y\) on \(D\), \(X\), and the first-stage
residual.
\end{frame}

\begin{frame}
\frametitle{Two step GMM: estimation and inference}
\begin{itemize}
  \item For first stage problem, we do OLS regression of \(D_{i}\)
    on \(S_{i}\).
    Moment condition is the usual OLS orthogonality condition.
    \vfill
  \item For second stage problem, we are doing a probit regression of
    \(Y_{i}\) on \(D_{i}\), \(X_{i}\) and \(\hat{\nu}_{n,i} = D_{i} - S_{i}'
    \hat{\delta}_{n}\).
    This is essentially an imputation of the conditional probit index:
    \begin{equation*}
      q_{i} (\alpha, \beta, \delta, \tau, \rho) = \frac{\alpha}{\sqrt{1 -
      \rho^{2}}} D_{i} + X_{i}' \frac{\beta}{\sqrt{1 - \rho^{2}}} -
      \frac{\rho}{\sqrt{\tau (1 - \rho^{2})}} (D_{i} - S_{i}' \delta).
    \end{equation*}
    \vfill
  \item The second stage problem also has a moment condition: we are setting the
    average probit conditional likelihood score equal to zero.
    \vfill
  \item Two ways to estimate and do inference:
    \vfill
    \begin{enumerate}
      \item Do first stage and then second stage separately.
        Use corrected standard errors based on stacked GMM.
        \vfill
      \item Solve the stacked GMM problem.
        Use the stacked GMM standard errors directly.
        \vfill
    \end{enumerate}
  \item Alternatively, use bootstrap.
\end{itemize}
\end{frame}

\begin{frame}{Two-step estimation: a linear probit index}
Write the same conditional index in linear form:
\begin{equation*}
  q = a D + X' b + c (D - S' \delta), \quad
  a = \frac{\alpha}{\sqrt{1 - \rho^{2}}}, \quad
  b = \frac{\beta}{\sqrt{1 - \rho^{2}}}, \quad
  c = -\frac{\rho}{\sqrt{\tau (1 - \rho^{2})}}.
\end{equation*}
\vfill
\begin{enumerate}
  \item Regress \(D_{i}\) on \(S_{i}\) by OLS to obtain
    \(\hat{\delta}_{n}\), and compute
    \begin{equation*}
      \hat{\nu}_{n,i} = D_{i} - S_{i}' \hat{\delta}_{n}, \quad
      \hat{\tau}_{n} = \frac{1}{n} \sum_{i = 1}^{n} \hat{\nu}_{n,i}^{2}.
    \end{equation*}
    \vfill
  \item Run an ordinary probit of \(Y_{i}\) on \(D_{i}\), \(X_{i}\), and
    \(\hat{\nu}_{n,i}\) to obtain \((\hat{a}_{n}, \hat{b}_{n}',
    \hat{c}_{n})'\).
    \vfill
  \item Transform these coefficients back to the structural parameters.
\end{enumerate}
%
% \vfill
%
% For \(\tau > 0\), the coefficient on the residual identifies \(\rho\):
% \begin{equation*}
%   c^{2} \tau = \frac{\rho^{2}}{1 - \rho^{2}}, \quad
%   \rho = -\frac{c \sqrt{\tau}}{\sqrt{1 + c^{2} \tau}}, \quad
%   \sqrt{1 - \rho^{2}} = \frac{1}{\sqrt{1 + c^{2} \tau}}.
% \end{equation*}
\vfill
The estimates of interest are
\begin{equation*}
  \hat{\alpha}_{n} = \frac{\hat{a}_{n}}
  {\sqrt{1 + \hat{c}_{n}^{2} \hat{\tau}_{n}}}, \quad
  \hat{\beta}_{n} = \frac{\hat{b}_{n}}
  {\sqrt{1 + \hat{c}_{n}^{2} \hat{\tau}_{n}}}.
\end{equation*}
\end{frame}

\begin{frame}{Two-step GMM: stacked moment condition}
Group the parameters as \(\eta = (\delta', \tau)'\) and
\(\theta = (a, b', c)'\).
\begin{equation*}
  g (\xi, \eta, \theta) =
  \begin{pmatrix} g_{1} (\xi, \eta) \\ g_{2} (\xi, \eta, \theta) \end{pmatrix}
  = \begin{pmatrix}
    S (D - S' \delta) \\
    (D - S' \delta)^{2} - \tau \\
    r \nabla_{\theta} q
  \end{pmatrix}, \quad
  \nabla_{\theta} q = \begin{pmatrix} D \\ X \\ D - S' \delta \end{pmatrix}.
\end{equation*}
\vfill
At the true parameter values,
\begin{equation*}
  \E [g (\xi, \eta_{0}, \theta_{0})] = \mathbf{0}_{2 k + l + 3}.
\end{equation*}
Here \(g_{1}\) has \(k + l + 1\) entries and \(g_{2}\) has \(k + 2\).
These are exactly identified moments.
\end{frame}

\begin{frame}{Two-step GMM: derivative of the moments}
Order derivative columns as \((\eta', \theta')\).
\begin{equation*}
  \frac{\partial g}{\partial (\eta', \theta')} =
  \begin{pmatrix}
    \frac{\partial g_{1}}{\partial \eta'} & \mathbf{0} \\
    \frac{\partial g_{2}}{\partial \eta'} &
    \frac{\partial g_{2}}{\partial \theta'}
  \end{pmatrix}, \quad
  \frac{\partial g_{1}}{\partial \eta'} =
  \begin{pmatrix}
    -S S' & \mathbf{0}_{k + l} \\
    -2 (D - S' \delta) S' & -1
  \end{pmatrix}.
\end{equation*}
\vfill
Since \(\partial_{q} r = -r (q + r)\) and \(\nabla_{\theta}^{2} q = 0\),
\begin{equation*}
  \begin{split}
    \frac{\partial g_{2}}{\partial \eta'} =
    & \, -r (q + r) (\nabla_{\theta} q) (\nabla_{\eta} q)'
    + r \frac{\partial (\nabla_{\theta} q)}{\partial \eta'}, \\
    \frac{\partial g_{2}}{\partial \theta'} =
    & \, -r (q + r) (\nabla_{\theta} q) (\nabla_{\theta} q)'.
  \end{split}
\end{equation*}
Second-stage derivative wrt \(\tau\) is zero.
Derivative wrt \(\delta\) accounts for generated residual.
\end{frame}

\begin{frame}{Two-step GMM: dependence on the estimated residual}
For the linear index \(q = a D + X' b + c (D - S' \delta)\),
\begin{equation*}
  \nabla_{\eta} q = \begin{pmatrix} -c S \\ 0 \end{pmatrix}, \quad
  \frac{\partial (\nabla_{\theta} q)}{\partial \eta'} =
  \begin{pmatrix}
    \mathbf{0}_{k + l}' & 0 \\
    \mathbf{0}_{k \times (k + l)} & \mathbf{0}_{k} \\
    -S' & 0
  \end{pmatrix}.
\end{equation*}
\vfill
Therefore,
\begin{equation*}
  \frac{\partial g_{2}}{\partial \delta'} =
  c r (q + r)
  \begin{pmatrix} D \\ X \\ D - S' \delta \end{pmatrix} S'
  - r \begin{pmatrix}
    \mathbf{0}_{k + l}' \\
    \mathbf{0}_{k \times (k + l)} \\
    S'
  \end{pmatrix}, \quad
  \frac{\partial g_{2}}{\partial \tau} = \mathbf{0}_{k + 2}.
\end{equation*}
\end{frame}

\begin{frame}{Two-step GMM: population derivative}
Denote the three derivative blocks by
\begin{equation*}
  A = \E \left[\frac{\partial g_{1}}{\partial \eta'}\right], \quad
  B = \E \left[\frac{\partial g_{2}}{\partial \eta'}\right], \quad
  C = \E \left[\frac{\partial g_{2}}{\partial \theta'}\right].
\end{equation*}
Since \(\E [S \nu] = \mathbf{0}\) and \(\E [r \mid D, S] = 0\),
\begin{equation*}
  \begin{gathered}
    A = \begin{pmatrix} -\E [S S'] & \mathbf{0} \\
      \mathbf{0}' & -1 \end{pmatrix}, \quad
    B = \left(c_{0} \E [r (q + r) (\nabla_{\theta} q) S'],
    \mathbf{0}_{k + 2}\right), \\
    C = -\E [r (q + r) (\nabla_{\theta} q) (\nabla_{\theta} q)'].
  \end{gathered}
\end{equation*}
\end{frame}

\begin{frame}{Two-step estimation: first-stage correction}
By Taylor expansion,
\begin{equation*}
  \begin{split}
    \sqrt{n} (\hat{\eta}_{n} - \eta_{0}) =
    & \, -A^{-1} \frac{1}{\sqrt{n}} \sum_{i = 1}^{n}
    g_{1} (\xi_{i}, \eta_{0}) + \op (1), \\
    \sqrt{n} (\hat{\theta}_{n} - \theta_{0}) =
    & \, -C^{-1} \frac{1}{\sqrt{n}} \sum_{i = 1}^{n}
    \left\{g_{2} (\xi_{i}, \eta_{0}, \theta_{0})
    - B A^{-1} g_{1} (\xi_{i}, \eta_{0})\right\} + \op (1).
  \end{split}
\end{equation*}
\vfill
Thus, asymptotic variance for linear-index coefficients is
\begin{equation*}
  V_{\theta} = C^{-1} \Var [g_{2} - B A^{-1} g_{1}] (C^{-1})'.
\end{equation*}
\vfill
This variance concerns \((\hat{a}_{n}, \hat{b}_{n}', \hat{c}_{n})'\).
Standard errors for the coefficients of interest,
\((\hat{\alpha}_{n}, \hat{\beta}_{n}')'\), require the delta method as well.
\end{frame}

\begin{frame}{Two-step estimation: variance under the Gaussian model}
At the truth, \(\E [g_{2} \mid D, S] = \mathbf{0}\), so
\(\Cov (g_{1}, g_{2}) = \mathbf{0}\), and \(\Var (g_{2}) = -C\).
The first-stage covariance is
\begin{equation*}
  A^{-1} \Var (g_{1}) (A^{-1})' =
  \begin{pmatrix}
    \tau_{0} (\E [S S'])^{-1} & \mathbf{0} \\
    \mathbf{0}' & 2 \tau_{0}^{2}
  \end{pmatrix}.
\end{equation*}
Hence
\begin{equation*}
  V_{\theta} = -C^{-1} + C^{-1} B
  \begin{pmatrix}
    \tau_{0} (\E [S S'])^{-1} & \mathbf{0} \\
    \mathbf{0}' & 2 \tau_{0}^{2}
  \end{pmatrix} B' (C^{-1})'.
\end{equation*}
\vfill
The last column of \(B\) is zero, and the first-stage covariance is block
diagonal.
Therefore the limiting covariance between \(\hat{\tau}_{n}\) and
\(\hat{\theta}_{n}\) is zero:
\begin{equation*}
  \sqrt{n} \begin{pmatrix}
    \hat{\tau}_{n} - \tau_{0} \\ \hat{\theta}_{n} - \theta_{0}
  \end{pmatrix} \dto \Normal \left(\mathbf{0},
  \begin{pmatrix} 2 \tau_{0}^{2} & \mathbf{0}' \\
    \mathbf{0} & V_{\theta} \end{pmatrix}\right).
\end{equation*}
\end{frame}

\begin{frame}{Main-equation coefficients: the delta method}
Differentiate the coefficients of interest \((\alpha, \beta')'\) with respect to
\((\tau, a, b', c)\):
\begin{equation*}
  J = \frac{\partial (\alpha, \beta')'}
  {\partial (\tau, a, b', c)} = \frac{1}{(1 + c^{2} \tau)^{3/2}}
  \begin{pmatrix}
    -a c^{2} / 2 & 1 + c^{2} \tau & \mathbf{0}_{k}' & -a c \tau \\
    -b c^{2} / 2 & \mathbf{0}_{k} & (1 + c^{2} \tau) I_{k} & -b c \tau
  \end{pmatrix}.
\end{equation*}
Here \(I_{k}\) is the \(k \times k\) identity matrix.
\vfill
Let \(J_{0}\) be this matrix at \((\tau_{0}, a_{0}, b_{0}', c_{0})\).
The asymptotic variance of \(\sqrt{n} ((\hat{\alpha}_{n},
\hat{\beta}_{n}')' - (\alpha_{0}, \beta_{0}')')\) is
\begin{equation*}
  J_{0} \begin{pmatrix} 2 \tau_{0}^{2} & \mathbf{0}' \\
    \mathbf{0} & V_{\theta} \end{pmatrix} J_{0}'.
\end{equation*}
First diagonal entry gives asymptotic variance for \(\alpha\);
the remaining block is for \(\beta\).
\end{frame}

\begin{frame}{Main-equation coefficients: feasible standard errors}
For a sample sandwich, let \(\hat{A}\), \(\hat{B}\), and \(\hat{C}\) be
sample averages of the derivative blocks at \((\hat{\eta}_{n},
\hat{\theta}_{n})\).
Estimate joint asymptotic covariance by
\begin{equation*}
  \hat{V} =
  \begin{pmatrix} \hat{A} & \mathbf{0} \\
    \hat{B} & \hat{C} \end{pmatrix}^{-1}
  \left[\frac{1}{n} \sum_{i = 1}^{n}
    g (\xi_{i}, \hat{\eta}_{n}, \hat{\theta}_{n})
    g (\xi_{i}, \hat{\eta}_{n}, \hat{\theta}_{n})'\right]
  \left[\begin{pmatrix} \hat{A} & \mathbf{0} \\
    \hat{B} & \hat{C} \end{pmatrix}^{-1}\right]'.
\end{equation*}
\vfill
Extract submatrix for \((\tau, a, b', c)'\), denoted
\(\hat{V}_{(\tau,\theta)}\), and evaluate \(J\) at the estimates.
The estimated covariance of \((\hat{\alpha}_{n}, \hat{\beta}_{n}')'\) is
\begin{equation*}
  \frac{1}{n} \hat{J} \hat{V}_{(\tau,\theta)} \hat{J}'.
\end{equation*}
Standard error for \(\hat{\alpha}_{n}\) is the square root of first
diagonal entry; remaining diagonal entries give those for \(\hat{\beta}_{n}\).
\end{frame}
